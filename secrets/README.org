#+Author: Kayvan Kazmeinajd
#+Email: kayvan@q2io.com


** Generate Private key
#+begin_src shell
nix shell nixpkgs#age -c age-keygen -o ~/.config/sops/age/keys.txt
#+end_src

** Generate Public key
#+begin_src shell
nix shell nixpkgs#age -c age-keygen -y ~/.config/sops/age/keys.txt
#+end_src

** create the [../.sops.yaml][.sops.yaml]] file

** create the secret file
#+begin_src  shell
cd ~/.config/dotfiles/modules/system/secrets
sop secrets.yaml
#+end_src
